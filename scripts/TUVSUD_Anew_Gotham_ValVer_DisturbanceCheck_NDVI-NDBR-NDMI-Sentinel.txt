MAX_CLOUD = 20

# ----- Cloud mask processor
def mask_s2_clouds(image):
    qa = image.select("QA60")
    cloud_bit_mask = 1 << 10
    cirrus_bit_mask = 1 << 11
    mask = qa.bitwiseAnd(cloud_bit_mask).eq(0).And(
           qa.bitwiseAnd(cirrus_bit_mask).eq(0))
    return image.updateMask(mask).divide(10000).copyProperties(image, ["system:time_start"])

# ----- Assemble pre and post composites
pre_composite = (ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")
    .filterDate("2023-10-18", "2024-01-18")
    .filterBounds(roi)
    .filter(ee.Filter.lt("CLOUDY_PIXEL_PERCENTAGE", MAX_CLOUD))
    .map(mask_s2_clouds)
    .median()
    .clip(roi))

post_composite = (ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")
    .filterDate("2024-09-30", "2024-12-31")
    .filterBounds(roi)
    .filter(ee.Filter.lt("CLOUDY_PIXEL_PERCENTAGE", MAX_CLOUD))
    .map(mask_s2_clouds)
    .median()
    .clip(roi))

# ----- Spectral index functions
# B2=Blue, B3=Green, B4=Red, B8=NIR, B8A=narNIR, B11=SWIR1, B12=SWIR2
def compute_ndvi(image):
    return image.normalizedDifference(["B8",  "B4" ]).rename("NDVI")

def compute_ndmi(image):
    return image.normalizedDifference(["B8",  "B11"]).rename("NDMI")

def compute_ndbr(image):
    return image.normalizedDifference(["B8A", "B12"]).rename("NDBR")

# ----- Single-date index images
pre_ndvi  = compute_ndvi(pre_composite)
post_ndvi = compute_ndvi(post_composite)
pre_ndmi  = compute_ndmi(pre_composite)
post_ndmi = compute_ndmi(post_composite)
pre_ndbr  = compute_ndbr(pre_composite)
post_ndbr = compute_ndbr(post_composite)

# ----- Multi-date difference images (positive = loss/deterioration)
delta_ndvi = pre_ndvi.subtract(post_ndvi).rename("dNDVI")
delta_ndmi = pre_ndmi.subtract(post_ndmi).rename("dNDMI")
delta_ndbr = pre_ndbr.subtract(post_ndbr).rename("dNBR")

# ----- Classification functions
# NDVI single-date: Water / Rock+Bare / Shrub+Grass / Forest
def classify_ndvi(ndvi_image):
    return (ndvi_image
            .where(ndvi_image.lt(-0.1), 0)
            .where(ndvi_image.gte(-0.1).And(ndvi_image.lt(0.2)), 1)
            .where(ndvi_image.gte(0.2).And(ndvi_image.lt(0.4)), 2)
            .where(ndvi_image.gte(0.4), 3)
            .rename("NDVI_class").toInt16())

# NDMI single-date: Bare / High stress / Average / Healthy
def classify_ndmi(ndmi_image):
    return (ndmi_image
            .where(ndmi_image.lt(-0.2), 0)
            .where(ndmi_image.gte(-0.2).And(ndmi_image.lt(0.0)), 1)
            .where(ndmi_image.gte(0.0).And(ndmi_image.lt(0.4)), 2)
            .where(ndmi_image.gte(0.4), 3)
            .rename("NDMI_class").toInt16())

# dNBR burn severity: Regrowth / Low / Moderate / High
def classify_ndbr(ndbr_diff_image):
    return (ndbr_diff_image
            .where(ndbr_diff_image.lt(-0.4), 0)
            .where(ndbr_diff_image.gte(-0.4).And(ndbr_diff_image.lt(-0.2)), 1)
            .where(ndbr_diff_image.gte(-0.2).And(ndbr_diff_image.lt(0.0)), 2)
            .where(ndbr_diff_image.gte(0.0), 3)
            .rename("NDBR_class").toInt16())

# dNDVI canopy loss: None / Low / Moderate / High
def classify_delta_ndvi(delta):
    return (delta
            .where(delta.lt(0.10), 0)
            .where(delta.gte(0.10).And(delta.lt(0.20)), 1)
            .where(delta.gte(0.20).And(delta.lt(0.35)), 2)
            .where(delta.gte(0.35), 3)
            .rename("dNDVI_class").toInt16())
