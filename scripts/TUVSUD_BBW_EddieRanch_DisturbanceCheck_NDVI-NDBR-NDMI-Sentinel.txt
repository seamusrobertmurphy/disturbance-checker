#  Output

#  dNDVI  — Canopy loss         (harvest, blowdown, deforestation)
#  dNDMI  — Moisture stress     (pest, drought, beetle damage)
#  dNBR   — Burn severity       (wildfire, prescribed burn)

#  Spectral Equations

#  NDVI  = (B8  - B4)  / (B8  + B4)    NIR vs Red          vegetation vigour
#  NDMI  = (B8  - B11) / (B8  + B11)   NIR vs SWIR1        canopy moisture
#  NBR   = (B8A - B12) / (B8A + B12)   narNIR vs SWIR2     fire / char signal
#

#  Change Trends

#  dNDVI = pre - post   positive = canopy loss
#  dNDMI = pre - post   positive = moisture loss / stress
#  dNBR  = post - pre   positive = burned area (MTBS convention)

#  Severity CLasses

#  dNDVI  Low (0.10-0.20)  |  Mod (0.20-0.35)  |  High (>0.35)
#  dNDMI  Low (0.15-0.30)  |  Mod (0.30-0.45)  |  High (>0.45)
#  dNBR   Low (0.10-0.27)  |  Mod (0.27-0.44)  |  High (>0.44)

#  Temporal Windows

#  Oct-Dec composites capture dormant/senescent canopy which inflates dNDMI.
#  Preferred windows for moisture stress detection: July-August (peak growing season).
#  Adjust RP_Start / RP_End below before running. This is the trade off we must weigh up

# --- Spatial filter: Blackfeet project area (WGS84 bounding box) ---
# Corners (lat, lon):
#   NW  39.54838,-123.45348
#   SW  39.13116,-123.37944
#   SE  39.13280,-122.72467

roi = ee.Geometry.Rectangle([-123.45348, 39.13116, -122.72467, 39.55327])
m.set_center(39.34253,-123.11394, 7)

# --- Temporal filter ---
# RECOMMENDED: switch to July-Aug for growing-season moisture stress detection.
# RP_START and RP_END windows allow normalization sampling.
# Keep an eye out for dormant-season senescence that inflates dNDMI extent (i.e. some RP dates fall in dead of winter)
RP_START_BEGIN = "2019-08-01"
RP_START_END   = "2019-09-30"
RP_END_BEGIN   = "2020-08-01"
RP_END_END     = "2020-09-30"

# --- Cloud filter ---
MAX_CLOUD = 30   # % cloudy pixels per scene; 30 is tighter than original 40


# Step IV: Median-Normalized Composites
# ---------------------------------

def mask_s2_clouds(image):
    """Remove cloud and cirrus pixels using Sentinel-2 QA60 bitmask band.
    Scales reflectance from DN (0-10000) to surface reflectance (0.0-1.0)."""
    qa              = image.select("QA60")
    cloud_bit_mask  = 1 << 10
    cirrus_bit_mask = 1 << 11
    mask = (qa.bitwiseAnd(cloud_bit_mask).eq(0)
              .And(qa.bitwiseAnd(cirrus_bit_mask).eq(0)))
    return (image.updateMask(mask)
                 .divide(10000)
                 .copyProperties(image, ["system:time_start"]))


def build_composite(start, end, roi_geom):
    """Filter S2 SR collection by date, bounds, and cloud cover,
    apply cloud mask, and return a median composite clipped to ROI."""
    return (ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")
              .filterDate(start, end)
              .filterBounds(roi_geom)
              .filter(ee.Filter.lt("CLOUDY_PIXEL_PERCENTAGE", MAX_CLOUD))
              .map(mask_s2_clouds)
              .median()
              .clip(roi_geom))


pre_composite  = build_composite(RP_START_BEGIN, RP_START_END, roi)
post_composite = build_composite(RP_END_BEGIN,   RP_END_END,   roi)


# Step V: Water Masking
# -----------------------

# Mask pixels with >= 50% water occurrence frequency to exclude lakes, rivers,
# and wetlands from disturbance analysis. These pixels produce spurious spectral
# change signals unrelated to forest condition.

water_mask = (ee.Image("JRC/GSW1_4/GlobalSurfaceWater")
                .select("occurrence")
                .lt(50)
                .unmask(1))

# ------------------------------------------------
# Step VI: Spectral Indices
# ------------------------------------------------

def compute_ndvi(image):
    """NDVI = (B8 - B4) / (B8 + B4)  |  NIR vs Red  |  vegetation vigour"""
    return image.normalizedDifference(["B8", "B4"]).rename("NDVI")


def compute_ndmi(image):
    """NDMI = (B8 - B11) / (B8 + B11)  |  NIR vs SWIR1  |  canopy moisture"""
    return image.normalizedDifference(["B8", "B11"]).rename("NDMI")


def compute_nbr(image):
    """NBR = (B8A - B12) / (B8A + B12)  |  narNIR vs SWIR2  |  fire/char signal"""
    return image.normalizedDifference(["B8A", "B12"]).rename("NBR")


# --- Single-date index images ---
pre_ndvi  = compute_ndvi(pre_composite)
post_ndvi = compute_ndvi(post_composite)
pre_ndmi  = compute_ndmi(pre_composite)
post_ndmi = compute_ndmi(post_composite)
pre_nbr   = compute_nbr(pre_composite)
post_nbr  = compute_nbr(post_composite)

# --- Multi-date difference images ---
# dNDVI = pre - post  (positive = canopy loss)
# dNDMI = pre - post  (positive = moisture stress / loss)
# dNBR  = post - pre  (positive = burned; MTBS convention)
delta_ndvi = (pre_ndvi.subtract(post_ndvi)
                       .rename("dNDVI")
                       .updateMask(water_mask))

delta_ndmi = (pre_ndmi.subtract(post_ndmi)
                       .rename("dNDMI")
                       .updateMask(water_mask))

delta_nbr  = (post_nbr.subtract(pre_nbr)
                       .rename("dNBR")
                       .updateMask(water_mask))

# Inspect the dNDMI distribution BEFORE committing to thresholds.
# A unimodal distribution centred near 0 = mostly noise / phenology.
# A bimodal distribution = genuine disturbance signal present.

    maxPixels= 1e9

# ------------------------------------------------
# Step VIII: Severity Classes
# ------------------------------------------------

# dNDVI canopy loss — harvest/deforestation detection (SOP Step 4A)
# Class 1 = Low-Moderate   (0.10 – 0.20)
# Class 2 = Moderate-High  (0.20 – 0.35)
# Class 3 = High           (> 0.35)
def classify_delta_ndvi(delta):
    return (delta
            .where(delta.lt(0.15), 0)
            .where(delta.gte(0.15).And(delta.lt(0.20)), 1)
            .where(delta.gte(0.20).And(delta.lt(0.30)), 2)
            .where(delta.gte(0.30), 3)
            .rename("dNDVI_class").toInt16())

# dNDMI moisture stress — pest/beetle/drought detection (SOP Step 3C)
# Class 1 = Emerging stress  (0.05 – 0.15)
# Class 2 = Moderate stress  (0.15 – 0.25)
# Class 3 = High stress      (> 0.25)
def classify_delta_ndmi(delta):
    return (delta
            .where(delta.lt(0.05), 0)
            .where(delta.gte(0.05).And(delta.lt(0.15)), 1)
            .where(delta.gte(0.15).And(delta.lt(0.25)), 2)
            .where(delta.gte(0.25), 3)
            .rename("dNDMI_class").toInt16())

# dNBR burn severity — MTBS/USFS PNW thresholds (post minus pre)
# Class 1 = Low severity      (0.10 – 0.27)
# Class 2 = Moderate severity (0.27 – 0.44)
# Class 3 = High severity     (> 0.44)
def classify_ndbr(delta):
    return (delta
            .where(delta.lt(0.10), 0)
            .where(delta.gte(0.10).And(delta.lt(0.27)), 1)
            .where(delta.gte(0.27).And(delta.lt(0.44)), 2)
            .where(delta.gte(0.44), 3)
            .rename("NDBR_class").toInt16())



# --- Apply classifiers ---
delta_ndvi_class = classify_delta_ndvi(delta_ndvi)
delta_ndmi_class = classify_delta_ndmi(delta_ndmi)
delta_nbr_class  = classify_ndbr(delta_nbr)

# --- Mask class 0 (undisturbed pixels → transparent on map) ---
delta_ndvi_disturbed = delta_ndvi_class.updateMask(delta_ndvi_class.gt(0))
delta_ndmi_disturbed = delta_ndmi_class.updateMask(delta_ndmi_class.gt(0))
delta_nbr_disturbed  = delta_nbr_class.updateMask(delta_nbr_class.gt(0))

# --------------------------------------------------
# Method Notes:
# --------------------------------------------------

# Patchy Composites
# If median composites show missing tiles or patchwork artefacts, widen the
# temporal window (filterDate) or relax MAX_CLOUD. The median normaliser
# requires a minimum pixel sample per cell — thin collections produce gaps.
# Note: most cloud/shadow gap-fill algorithms sample forward in time (earlier
# to later dates). Some sample backwards. This affects which season's spectral
# signal dominates filled pixels — a key concern for seasonally variable
# forested landscapes like the Blackfeet project area.

# Threshold Recalibration
# Always run the histogram block (Step VIII) before committing to thresholds.
# A good disturbance signal produces a bimodal histogram with a clear gap
# between the noise bulk (near 0) and the disturbance tail (> 0.15).
# If the histogram is unimodal, consider widening the temporal window or
# switching to growing-season composites (July-August).

# Band Naming
#  All delta image band names follow the d-prefix convention:
#   dNDVI → dNDVI_class
#   dNDMI → dNDMI_class
#   dNBR  → dNBR_class
# Use these exact strings when referencing result dict keys (e.g. result["dNDMI"]).


# ==================================================
# Step XI: Multi-Period Reporting Stack (RP1–RP6)
# ==================================================
#
# Builds disturbance layers for six consecutive annual reporting periods
# using the same compositing windows, classifiers, and thresholds defined
# above. Each RP uses Aug–Sep pre and Aug–Sep post composites to maintain
# growing-season consistency across years.
#
# RP1 duplicates the single-period analysis above; it is re-derived here
# so all six periods live in a single dict for batch export.

RP_DATES = {
    "RP1": {"pre_start": "2019-08-01", "pre_end": "2019-09-30",
            "post_start": "2020-08-01", "post_end": "2020-09-30"},
    "RP2": {"pre_start": "2020-08-01", "pre_end": "2020-09-30",
            "post_start": "2021-08-01", "post_end": "2021-09-30"},
    "RP3": {"pre_start": "2021-08-01", "pre_end": "2021-09-30",
            "post_start": "2022-08-01", "post_end": "2022-09-30"},
    "RP4": {"pre_start": "2022-08-01", "pre_end": "2022-09-30",
            "post_start": "2023-08-01", "post_end": "2023-09-30"},
    "RP5": {"pre_start": "2023-08-01", "pre_end": "2023-09-30",
            "post_start": "2024-08-01", "post_end": "2024-09-30"},
    "RP6": {"pre_start": "2024-08-01", "pre_end": "2024-09-30",
            "post_start": "2025-08-01", "post_end": "2025-09-30"},
}

rp_results = {}

for rp_name, dates in RP_DATES.items():

    # --- Composites ---
    rp_pre  = build_composite(dates["pre_start"],  dates["pre_end"],  roi)
    rp_post = build_composite(dates["post_start"], dates["post_end"], roi)

    # --- Spectral indices ---
    rp_pre_ndvi  = compute_ndvi(rp_pre)
    rp_post_ndvi = compute_ndvi(rp_post)
    rp_pre_ndmi  = compute_ndmi(rp_pre)
    rp_post_ndmi = compute_ndmi(rp_post)
    rp_pre_nbr   = compute_nbr(rp_pre)
    rp_post_nbr  = compute_nbr(rp_post)

    # --- Delta images (water-masked) ---
    rp_dndvi = (rp_pre_ndvi.subtract(rp_post_ndvi)
                .rename("dNDVI").updateMask(water_mask))
    rp_dndmi = (rp_pre_ndmi.subtract(rp_post_ndmi)
                .rename("dNDMI").updateMask(water_mask))
    rp_dnbr  = (rp_post_nbr.subtract(rp_pre_nbr)
                .rename("dNBR").updateMask(water_mask))

    # --- Classify and mask undisturbed ---
    rp_dndvi_cls = classify_delta_ndvi(rp_dndvi)
    rp_dndmi_cls = classify_delta_ndmi(rp_dndmi)
    rp_dnbr_cls  = classify_ndbr(rp_dnbr)

    rp_dndvi_dist = rp_dndvi_cls.updateMask(rp_dndvi_cls.gt(0))
    rp_dndmi_dist = rp_dndmi_cls.updateMask(rp_dndmi_cls.gt(0))
    rp_dnbr_dist  = rp_dnbr_cls.updateMask(rp_dnbr_cls.gt(0))

    # --- Store for export ---
    rp_results[rp_name] = {
        "pre_RGB":  rp_pre,
        "post_RGB": rp_post,
        "dNDVI":    rp_dndvi_dist,
        "dNDMI":    rp_dndmi_dist,
        "dNBR":     rp_dnbr_dist,
    }

    # --- Add classified layers to map ---
    show = (rp_name != "RP1")  # RP1 already rendered above
    year_label = f"{dates['pre_start'][:4]}–{dates['post_start'][:4]}"

    m.addLayer(rp_dndvi_dist, vis_dndvi,
               f"{rp_name} dNDVI Canopy Loss     [{year_label}]", shown=show)
    m.addLayer(rp_dndmi_dist, vis_dndmi,
               f"{rp_name} dNDMI Moisture Stress [{year_label}]", shown=show)
    m.addLayer(rp_dnbr_dist,  vis_dnbr,
               f"{rp_name} dNBR  Burn Severity   [{year_label}]", shown=show)

    print(f"{rp_name} ({year_label}) — layers added to map")

# Step XII: Cumulative Disturbance Stack
# ------------------------------------------------

# Maximum-severity images across all six RPs per index. Each pixel records

all_dndvi = [rp_results[rp]["dNDVI"] for rp in RP_DATES]
all_dndmi = [rp_results[rp]["dNDMI"] for rp in RP_DATES]

cumulative_dnbr  = (ee.ImageCollection(all_dnbr).max()
                    .updateMask(ee.ImageCollection(all_dnbr).max().gt(0)))
