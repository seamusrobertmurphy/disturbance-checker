
# ----- Activate Earth Engine
ee.Authenticate()
ee.Initialize(project="murphys-deforisk")

# ----- Define Window & Centroid 
# Clockwise from top left corner => NW, SW, SE, NE
#NW = 44.5505,-91.0084
#SW = 44.1153,-90.9606
#SE = 44.1143,-90.1836
#NE = 44.5985,-90.1808

roi = ee.Geometry.Rectangle([-91.0084, 44.1153, -90.1836, 44.5985])
m.set_center(-118.393, 48.562, 8)



# ----- Temporal & Cloud Filter
MAX_CLOUD = 10

# ----- Cloud mask processor
def mask_s2_clouds(image):
    qa = image.select("QA60")
    cloud_bit_mask = 1 << 10
    cirrus_bit_mask = 1 << 11
    mask = qa.bitwiseAnd(cloud_bit_mask).eq(0).And(
           qa.bitwiseAnd(cirrus_bit_mask).eq(0))
    return image.updateMask(mask).divide(10000).copyProperties(image, ["system:time_start"])

# ----- Build composites (single ROI)
def build_composite(start, end, roi_geom):
    return (ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")
        .filterDate(start, end)
        .filterBounds(roi_geom)
        .filter(ee.Filter.lt("CLOUDY_PIXEL_PERCENTAGE", MAX_CLOUD))
        .map(mask_s2_clouds)
        .median()
        .clip(roi_geom))

# Temporal Window - reporting period
pre_composite  = build_composite("2023-03-01", "2023-05-01", roi)
post_composite = build_composite("2024-03-01", "2024-05-01", roi)


# ----- Water mask — exclude permanent/seasonal water bodies
# JRC Global Surface Water — mask pixels with >50% water occurrence
water_mask = (ee.Image("JRC/GSW1_4/GlobalSurfaceWater")
              .select("occurrence")
              .lt(50)
              .unmask(1))

# ----- Spectral index functions
# B2=Blue B3=Green B4=Red B8=NIR B8A=narNIR B11=SWIR1 B12=SWIR2
def compute_ndvi(image):
    return image.normalizedDifference(["B8",  "B4" ]).rename("NDVI")
def compute_ndmi(image):
    return image.normalizedDifference(["B8",  "B11"]).rename("NDMI")
def compute_ndbr(image):
    return image.normalizedDifference(["B8A", "B12"]).rename("NDBR")

# ----- Single-date indices
pre_ndvi  = compute_ndvi(pre_composite)
post_ndvi = compute_ndvi(post_composite)
pre_ndmi  = compute_ndmi(pre_composite)
post_ndmi = compute_ndmi(post_composite)
pre_ndbr  = compute_ndbr(pre_composite)
post_ndbr = compute_ndbr(post_composite)

# ----- Multi-date difference images
# dNDVI/dNDMI: pre minus post (positive = loss/stress)
# dNBR: post minus pre (positive = burned, standard MTBS convention)
delta_ndvi = pre_ndvi.subtract(post_ndvi).rename("dNDVI")
delta_ndmi = pre_ndmi.subtract(post_ndmi).rename("dNDMI")
delta_ndbr = post_ndbr.subtract(pre_ndbr).rename("dNBR") 

# ----- Apply water mask to all difference images
delta_ndvi = delta_ndvi.updateMask(water_mask)
delta_ndmi = delta_ndmi.updateMask(water_mask)
delta_ndbr = delta_ndbr.updateMask(water_mask)


# ----- Classification functions

# dNDVI canopy loss — harvest/deforestation detection (SOP Step 4A)
# Class 1 = Low-Moderate   (0.10 – 0.20)
# Class 2 = Moderate-High  (0.20 – 0.35)
# Class 3 = High           (> 0.35)
def classify_delta_ndvi(delta):
    return (delta
            .where(delta.lt(0.10), 0)
            .where(delta.gte(0.10).And(delta.lt(0.20)), 1)
            .where(delta.gte(0.20).And(delta.lt(0.35)), 2)
            .where(delta.gte(0.35), 3)
            .rename("dNDVI_class").toInt16())

# dNDMI moisture stress — pest/beetle/drought detection (SOP Step 3C)
# Class 1 = Emerging stress  (0.05 – 0.15)
# Class 2 = Moderate stress  (0.15 – 0.25)
# Class 3 = High stress      (> 0.25)
def classify_delta_ndmi(delta):
    return (delta
            .where(delta.lt(0.05), 0)
            .where(delta.gte(0.05).And(delta.lt(0.15)), 1)
            .where(delta.gte(0.15).And(delta.lt(0.25)), 2)
            .where(delta.gte(0.25), 3)
            .rename("dNDMI_class").toInt16())

# dNBR burn severity — MTBS/USFS PNW thresholds (post minus pre)
# Class 1 = Low severity      (0.10 – 0.27)
# Class 2 = Moderate severity (0.27 – 0.44)
# Class 3 = High severity     (> 0.44)
def classify_ndbr(delta):
    return (delta
            .where(delta.lt(0.10), 0)
            .where(delta.gte(0.10).And(delta.lt(0.27)), 1)
            .where(delta.gte(0.27).And(delta.lt(0.44)), 2)
            .where(delta.gte(0.44), 3)
            .rename("NDBR_class").toInt16())

# ----- Apply classifications
delta_ndvi_class = classify_delta_ndvi(delta_ndvi)
delta_ndmi_class = classify_delta_ndmi(delta_ndmi)
ndbr_class       = classify_ndbr(delta_ndbr)

# ----- Mask class 0 (undisturbed = transparent)
delta_ndvi_disturbed = delta_ndvi_class.updateMask(delta_ndvi_class.gt(0))
delta_ndmi_disturbed = delta_ndmi_class.updateMask(delta_ndmi_class.gt(0))
ndbr_disturbed       = ndbr_class.updateMask(ndbr_class.gt(0))


# ----- Visualisation parameters
vis_rgb = {"bands": ["B4", "B3", "B2"], "min": 0.02, "max": 0.25, "gamma": 1.2}
vis_ndvi   = {"min": -0.2, "max": 0.8, "palette": ["#d73027","#fc8d59","#fee08b","#d9ef8b","#1a9850"]}
vis_ndmi   = {"min": -0.5, "max": 0.5, "palette": ["#d73027","#fc8d59","#ffffbf","#91bfdb","#4575b4"]}
vis_delta  = {"min": -0.3, "max": 0.5, "palette": ["#1a9850","#ffffbf","#fc8d59","#d73027"]}

# 3-class severity palettes: yellow=low, orange=moderate, red=high
vis_dndvi  = {"min": 1, "max": 3, "palette": ["#FFEDA0", "#FC4E2A", "#800026"]}
vis_ndmi_d = {"min": 1, "max": 3, "palette": ["#FEB24C", "#FD8D3C", "#B10026"]}
vis_ndbr   = {"min": 1, "max": 3, "palette": ["#FFEDA0", "#FC4E2A", "#800026"]}  # ← matches dNDVI

# ----- Map layers
m.addLayer(pre_composite,  vis_rgb, "Pre RGB",  shown=False)
m.addLayer(post_composite, vis_rgb, "Post RGB", shown=False)

m.addLayer(pre_ndvi,  vis_ndvi, "Pre NDVI",  shown=False)
m.addLayer(post_ndvi, vis_ndvi, "Post NDVI", shown=False)
m.addLayer(pre_ndmi,  vis_ndmi, "Pre NDMI",  shown=False)
m.addLayer(post_ndmi, vis_ndmi, "Post NDMI", shown=False)

m.addLayer(delta_ndvi, vis_delta, "dNDVI continuous", shown=False)
m.addLayer(delta_ndmi, vis_delta, "dNDMI continuous", shown=False)
m.addLayer(delta_ndbr, vis_delta, "dNBR  continuous", shown=False)

# Classified disturbance layers (on by default)
m.addLayer(delta_ndvi_disturbed, vis_dndvi,  "dNDVI – Canopy Loss [Low/Mod/High]",     shown=True)
m.addLayer(delta_ndmi_disturbed, vis_ndmi_d, "dNDMI – Moisture Stress [Low/Mod/High]", shown=True)
m.addLayer(ndbr_disturbed,       vis_ndbr,   "dNBR  – Burn Severity [Low/Mod/High]",   shown=True)

# ROI boundary
m.addLayer(ee.FeatureCollection([ee.Feature(roi)]),
           {"color": "cyan", "fillColor": "00000000"}, "ROI")


# ----- Export to Google Driveccl
EXPORT_DRIVE = False

if EXPORT_DRIVE:
    export_layers = {
        "dNDVI_canopy_loss_classified":     delta_ndvi_disturbed,
        "dNDMI_moisture_stress_classified": delta_ndmi_disturbed,
        "dNBR_burn_severity_classified":    ndbr_disturbed,
        "Pre_RGB":                          pre_composite,
        "Post_RGB":                         post_composite,
    }
    for name, layer in export_layers.items():
        task = ee.batch.Export.image.toDrive(
            image=layer,
            description=name,
            folder="GEE_Exports",
            fileNamePrefix=name,
            scale=10,
            region=roi,
            crs="EPSG:3857",      
            maxPixels=1e13,
            fileFormat="GeoTIFF"
        )
        task.start()

# ----- INDEX REFERENCE
# NDVI  = (B8 - B4)   / (B8 + B4)      NIR, Red
# NDMI  = (B8 - B11)  / (B8 + B11)     NIR, SWIR1
# NBR   = (B8A - B12) / (B8A + B12)    narNIR, SWIR2
#
# dNDVI thresholds:
#   No loss:     < 0.10
#   Low-Mod:   0.10 – 0.20
#   Mod-High:  0.20 – 0.35
#   High:        > 0.35
#
# dNDMI thresholds):
#   No stress:   < 0.05
#   Emerging:  0.05 – 0.15
#   Moderate:  0.15 – 0.25
#   High:        > 0.25
#
# dNBR thresholds):
#   Regrowth:    < -0.40
#   Low:      -0.40 – -0.20
#   Mod-High: -0.20 –  0.00
#   High:        >  0.00
