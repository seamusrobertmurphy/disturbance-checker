#  Output

#  dNDVI  — Canopy loss         (harvest, blowdown, deforestation)
#  dNDMI  — Moisture stress     (pest, drought, beetle damage)
#  dNBR   — Burn severity       (wildfire, prescribed burn)

#  Spectral Equations

#  NDVI  = (B8  - B4)  / (B8  + B4)    NIR vs Red          vegetation vigour
#  NDMI  = (B8  - B11) / (B8  + B11)   NIR vs SWIR1        canopy moisture
#  NBR   = (B8A - B12) / (B8A + B12)   narNIR vs SWIR2     fire / char signal
#

#  Change Trends

#  dNDVI = pre - post   positive = canopy loss
#  dNDMI = pre - post   positive = moisture loss / stress
#  dNBR  = post - pre   positive = burned area (MTBS convention)

#  Severity CLasses

#  dNDVI  Low (0.10-0.20)  |  Mod (0.20-0.35)  |  High (>0.35)
#  dNDMI  Low (0.15-0.30)  |  Mod (0.30-0.45)  |  High (>0.45)   
#  dNBR   Low (0.10-0.27)  |  Mod (0.27-0.44)  |  High (>0.44)

#  Temporal Windows

#  Oct-Dec composites capture dormant/senescent canopy which inflates dNDMI.
#  Preferred windows for moisture stress detection: July-August (peak growing season).
#  Adjust RP_Start / RP_End below before running. This is the trade off we must weigh up

# --- Temporal filter ---

# RP_START and RP_END windows allow normalization sampling. 
# Keep an eye out for dormant-season senescence that inflates dNDMI extent (i.e. some RP dates fall in dead of winter)
RP_START_BEGIN = "2023-01-01"
RP_START_END   = "2023-02-28"
RP_END_BEGIN   = "2025-01-01"
RP_END_END     = "2025-02-28"

# --- Cloud filter ---
MAX_CLOUD = 30   # % cloudy pixels per scene; 30 is tighter than original 40


# Step IV: Median-Normalized Composites 
# ---------------------------------

def mask_s2_clouds(image):
    """Remove cloud and cirrus pixels using Sentinel-2 QA60 bitmask band.
    Scales reflectance from DN (0-10000) to surface reflectance (0.0-1.0)."""
    qa              = image.select("QA60")
    cloud_bit_mask  = 1 << 10
    cirrus_bit_mask = 1 << 11
    mask = (qa.bitwiseAnd(cloud_bit_mask).eq(0)
              .And(qa.bitwiseAnd(cirrus_bit_mask).eq(0)))
    return (image.updateMask(mask)
                 .divide(10000)
                 .copyProperties(image, ["system:time_start"]))


def build_composite(start, end, roi_geom):
    """Filter S2 SR collection by date, bounds, and cloud cover,
    apply cloud mask, and return a median composite clipped to ROI."""
    return (ee.ImageCollection("COPERNICUS/S2_SR_HARMONIZED")
              .filterDate(start, end)
              .filterBounds(roi_geom)
              .filter(ee.Filter.lt("CLOUDY_PIXEL_PERCENTAGE", MAX_CLOUD))
              .map(mask_s2_clouds)
              .median()
              .clip(roi_geom))


pre_composite  = build_composite(RP_START_BEGIN, RP_START_END, roi)
post_composite = build_composite(RP_END_BEGIN,   RP_END_END,   roi)


# Step V: Water Masking 
# -----------------------

# Mask pixels with >= 50% water occurrence frequency to exclude lakes, rivers,
# and wetlands from disturbance analysis. These pixels produce spurious spectral
# change signals unrelated to forest condition.

water_mask = (ee.Image("JRC/GSW1_4/GlobalSurfaceWater")
                .select("occurrence")
                .lt(50)
                .unmask(1))

# ------------------------------------------------
# Step VI: Spectral Indices
# ------------------------------------------------

def compute_ndvi(image):
    """NDVI = (B8 - B4) / (B8 + B4)  |  NIR vs Red  |  vegetation vigour"""
    return image.normalizedDifference(["B8", "B4"]).rename("NDVI")


def compute_ndmi(image):
    """NDMI = (B8 - B11) / (B8 + B11)  |  NIR vs SWIR1  |  canopy moisture"""
    return image.normalizedDifference(["B8", "B11"]).rename("NDMI")


def compute_nbr(image):
    """NBR = (B8A - B12) / (B8A + B12)  |  narNIR vs SWIR2  |  fire/char signal"""
    return image.normalizedDifference(["B8A", "B12"]).rename("NBR")


# --- Single-date index images ---
pre_ndvi  = compute_ndvi(pre_composite)
post_ndvi = compute_ndvi(post_composite)
pre_ndmi  = compute_ndmi(pre_composite)
post_ndmi = compute_ndmi(post_composite)
pre_nbr   = compute_nbr(pre_composite)
post_nbr  = compute_nbr(post_composite)

# --- Multi-date difference images ---
# dNDVI = pre - post  (positive = canopy loss)
# dNDMI = pre - post  (positive = moisture stress / loss)
# dNBR  = post - pre  (positive = burned; MTBS convention)
delta_ndvi = (pre_ndvi.subtract(post_ndvi)
                       .rename("dNDVI")
                       .updateMask(water_mask))

delta_ndmi = (pre_ndmi.subtract(post_ndmi)
                       .rename("dNDMI")
                       .updateMask(water_mask))

delta_nbr  = (post_nbr.subtract(pre_nbr)
                       .rename("dNBR")
                       .updateMask(water_mask))

def classify_delta_ndvi(delta):
    """Classify dNDVI into canopy loss severity classes.
    Class 0 = undisturbed (masked out of map display)
    Class 1 = Low-Moderate  (0.15 – 0.30)
    Class 2 = Moderate-High (0.30 – 0.45)
    Class 3 = High          (> 0.45)
    """
    return (delta
            .where(delta.lt(0.15), 0)
            .where(delta.gte(0.15).And(delta.lt(0.30)), 1)
            .where(delta.gte(0.30).And(delta.lt(0.45)), 2)
            .where(delta.gte(0.45), 3)
            .rename("dNDVI_class")
            .toInt16())


def classify_delta_ndmi(delta):
    """Classify dNDMI into moisture stress severity classes.
    Noise floor raised from 0.05 to 0.15 to reduce phenological false positives,
    particularly in Oct-Dec dormant-season composites.
    Class 0 = undisturbed / noise (masked out of map display)
    Class 1 = Low-Moderate  (0.15 – 0.30)
    Class 2 = Moderate-High (0.30 – 0.45)
    Class 3 = High          (> 0.45)
    """
    return (delta
            .where(delta.lt(0.15), 0)
            .where(delta.gte(0.15).And(delta.lt(0.30)), 1)
            .where(delta.gte(0.30).And(delta.lt(0.45)), 2)
            .where(delta.gte(0.45), 3)
            .rename("dNDMI_class")
            .toInt16())


def classify_delta_nbr(delta):
    """Classify dNBR into burn severity classes (MTBS/USFS PNW standard).
    Class 0 = unburned (masked out of map display)
    Class 1 = Low          (0.15 – 0.27)
    Class 2 = Moderate     (0.27 – 0.44)
    Class 3 = High         (> 0.44)
    """
    return (delta
            .where(delta.lt(0.15), 0)
            .where(delta.gte(0.15).And(delta.lt(0.27)), 1)
            .where(delta.gte(0.27).And(delta.lt(0.44)), 2)
            .where(delta.gte(0.44), 3)
            .rename("dNBR_class")    # FIXED: was "NDBR_class"
            .toInt16())


# --- Apply classifiers ---

# --------------------------------------------------
# Method Notes:
# --------------------------------------------------

# Patchy Composites
# If median composites show missing tiles or patchwork artefacts, widen the
# temporal window (filterDate) or relax MAX_CLOUD. The median normaliser
# requires a minimum pixel sample per cell — thin collections produce gaps.
# Note: most cloud/shadow gap-fill algorithms sample forward in time (earlier
# to later dates). Some sample backwards. This affects which season's spectral
# signal dominates filled pixels — a key concern for seasonally variable
# forested landscapes like the Blackfeet project area.

# Threshold Recalibration
# Always run the histogram block (Step VIII) before committing to thresholds.
# A good disturbance signal produces a bimodal histogram with a clear gap
# between the noise bulk (near 0) and the disturbance tail (> 0.15).
# If the histogram is unimodal, consider widening the temporal window or
# switching to growing-season composites (July-August).

# Band Naming
#  All delta image band names follow the d-prefix convention:
#   dNDVI → dNDVI_class
#   dNDMI → dNDMI_class
#   dNBR  → dNBR_class
# Use these exact strings when referencing result dict keys (e.g. result["dNDMI"]).
